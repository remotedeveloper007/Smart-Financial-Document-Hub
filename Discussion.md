# 7. Plan enforcement service

This is where restrictions happen.

```php id="bill6"
namespace App\Services\Billing;

use App\Models\Subscription;

class PlanLimitService
{
    public function check(int $teamId, string $metric, int $increment = 1): bool
    {
        $subscription = Subscription::with('plan')
            ->where('team_id', $teamId)
            ->first();

        $plan = $subscription->plan;

        $usage = $subscription->team->usageCounters()
            ->where('metric', $metric)
            ->where('period', now()->format('Y-m'))
            ->first();

        $used = $usage?->used ?? 0;

        $limit = match ($metric) {
            'documents_uploaded' => $plan->documents_limit,
            'ocr_pages' => $plan->ocr_pages_limit,
            'exports' => $plan->exports_limit,
            default => null,
        };

        if (is_null($limit)) {
            return true; // unlimited
        }

        return ($used + $increment) <= $limit;
    }
}
```

# 5. Make jobs idempotent (VERY IMPORTANT)

Update `ProcessDocumentJob`:

```php id="r5"
public function handle(ExtractTransactionsAction $extractAction): void
{
    $document = Document::findOrFail($this->documentId);

    // Prevent duplicate processing
    if ($document->status === 'processed') {
        return;
    }

    if ($document->status === 'processing') {
        return;
    }

    $document->update(['status' => 'processing']);

    try {
        $transactions = $extractAction->execute($document);

        app(\App\Actions\Transactions\StoreTransactionsAction::class)
            ->execute($document->team_id, $document->id, $transactions);

        $document->update(['status' => 'processed']);

    } catch (\Throwable $e) {

        $document->update(['status' => 'failed']);

        \App\Models\ProcessingFailure::create([
            'team_id' => $document->team_id,
            'document_id' => $document->id,
            'stage' => 'extraction',
            'error_message' => $e->getMessage(),
            'context' => [
                'file' => $document->file_path,
            ],
        ]);

        throw $e; // allow retry mechanism
    }
}
```

# 4. Matching Engine (core logic)

This is your **auto-reconciliation brain**.

```php id="rec4"
namespace App\Services\Reconciliation;

use App\Models\Transaction;
use App\Models\ExternalTransaction;
use App\Models\TransactionMatch;

class ReconciliationService
{
    public function match(int $teamId): void
    {
        $transactions = Transaction::where('team_id', $teamId)->get();
        $external = ExternalTransaction::where('team_id', $teamId)->get();

        foreach ($transactions as $tx) {
            foreach ($external as $ext) {

                $confidence = $this->calculateConfidence($tx, $ext);

                if ($confidence >= 85) {
                    TransactionMatch::updateOrCreate(
                        [
                            'transaction_id' => $tx->id,
                            'external_transaction_id' => $ext->id,
                        ],
                        [
                            'team_id' => $teamId,
                            'confidence' => $confidence,
                        ]
                    );
                }
            }
        }
    }

    private function calculateConfidence($tx, $ext): float
    {
        $score = 0;

        if (abs($tx->amount - $ext->amount) < 0.01) {
            $score += 50;
        }

        if ($tx->transaction_date == $ext->transaction_date) {
            $score += 30;
        }

        if (similar_text($tx->description, $ext->description) > 70) {
            $score += 20;
        }

        return $score;
    }
}
```

```php id="x4"
class ZohoExportService
{
    public function generate(array $data): array
    {
        return [
            'journal_entries' => array_map(function ($row) {
                return [
                    'date' => $row['date'],
                    'description' => $row['description'],
                    'reference_number' => $row['reference'],
                    'line_items' => [
                        [
                            'account_name' => $this->mapAccount($row['type']),
```

```php
class DocumentStateService
{
    private array $allowed = [
        'uploaded' => ['queued'],
        'queued' => ['processing'],
        'processing' => ['processed', 'failed'],
        'failed' => ['queued'],
        'processed' => []
    ];

    public function transition(Document $doc, string $to): void
    {
        if (!in_array($to, $this->allowed[$doc->status])) {
            throw new \Exception("Invalid state transition");
        }

        $doc->update(['status' => $to]);
    }
}
```

# 8. VERIFICATION SYSTEM (DETECT TAMPERING)

## 8.1 Integrity checker

```php id="audit_13"
class AuditIntegrityService
{
    public function verify(): bool
    {
        $logs = AuditLog::orderBy('id')->get();

        $previousHash = null;

        foreach ($logs as $log) {
            $expected = hash('sha256', json_encode([
                'team_id' => $log->team_id,
                'user_id' => $log->user_id,
                'action' => $log->action,
                'entity_type' => $log->entity_type,
                'entity_id' => $log->entity_id,
                'before' => $log->before,
                'after' => $log->after,
                'ip_address' => $log->ip_address,
            ]) . $previousHash);

            if ($log->hash !== $expected) {
                return false; // tampering detected
            }

            $previousHash = $log->hash;
        }

        return true;
    }
}
```
